  \seccionpill{\faIcon{robot}\ \ ¿Cómo uso la IA?}
  \vspace{0.15cm}

  {\footnotesize Genero el contenido de redes sociales de PROMEDIMSA combinando \textbf{dos
  asistentes de IA}, cada uno con un rol distinto:}

  \vspace{2pt}
  \begin{center}
  \begin{tabular}{@{}c c c@{}}
    \begin{tikzpicture}
      \node[circle, fill=cNavy, minimum size=1.0cm, inner sep=0pt]
        {\textcolor{white}{\small\faIcon{terminal}}};
    \end{tikzpicture}
    &
    \begin{tikzpicture}
      \node[fill=none] {\textcolor{cLimeDark}{\faIcon{arrow-right}}};
    \end{tikzpicture}
    &
    \begin{tikzpicture}
      \node[circle, fill=cNavy, minimum size=1.0cm, inner sep=0pt]
        {\textcolor{white}{\small\faIcon{image}}};
    \end{tikzpicture}
    \\
    {\scriptsize\bfseries Claude Code} & & {\scriptsize\bfseries ChatGPT}\\
    {\scriptsize redacta caption + prompt} & & {\scriptsize genera la imagen final}\\
  \end{tabular}
  \end{center}

  \vspace{1pt}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \itemcheck{1. Reviso lo ya publicado}{Consulto un índice de publicaciones anteriores
        para no repetir tema, título ni íconos.}
      \itemcheck{2. Uso datos reales}{Un archivo con los datos verificados de la empresa
        evita que la IA invente teléfonos, certificaciones o servicios.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \itemcheck{3. Genero caption + prompt}{Claude Code escribe el texto y el prompt de
        imagen siguiendo reglas fijas de tono y formato.}
      \itemcheck{4. Genero y publico}{ChatGPT crea la imagen sobre la plantilla de marca
        y yo la publico en redes.}
    \end{column}
  \end{columns}
\end{frame}
